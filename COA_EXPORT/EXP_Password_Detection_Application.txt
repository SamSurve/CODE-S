EXP — Password Detection Application
; EXP : Password Detection Application
org 100h

lea dx, prompt
mov ah, 09h
int 21h

lea dx, buf
mov ah, 0Ah
int 21h

lea si, pass
lea di, [buf+2]
mov cx, 5
check_loop:
    mov al, [si]
    mov bl, [di]
    cmp al, bl
    jne not_matched
    inc si
    inc di
    loop check_loop

mov al, [di]          ; next char must be Enter (0Dh)
cmp al, 0Dh
jne not_matched

lea dx, msg_matched
jmp print

not_matched:
lea dx, msg_unmatched

print:
mov ah, 09h
int 21h
hlt

prompt db 'Enter Password: $'
pass db 'TOMMY'
buf db 6, 0, 6 dup(0)
msg_matched db 0dh, 0ah, 'Password Matched$'
msg_unmatched db 0dh, 0ah, 'Password does not match$'

OUTPUT:
Enter Password: TOMMY

Password Matched
[CPU Halted (HLT)]
